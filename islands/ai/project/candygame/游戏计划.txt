const IMG = {

const game = document.getElementById("game");
let score = 0;

let player = create("player", IMG.player, 200, 200);
let shop = create("shop", IMG.shop, 600, 180);
let npc = create("npc", IMG.npc, 520, 220);

let candies = [];

for(let i=0;i<60;i++){
candies.push(create("candy", IMG.candy,

Math.random()*600));
}

/* 皇冠 */

let keys = {};

/* =========================
   创建对象
========================= */
function create(cls,img,x,y){
let el=document.createElement("div");
el.className=cls;
el.style.backgroundImage=`url(${img})`;
el.style.left=x+"px";
el.style.top=y+"px";
game.appendChild(el);
return el;
}

/* =========================
   主循环
========================= */
function loop(){

let x=parseInt(player.style.left);
let y=parseInt(player.style.top);

/* 移动 */
if(keys["w"]) y-=speed;
if(keys["s"]) y+=speed;
if(keys["a"]) x-=speed;
if(keys["d"]) x+=speed;

player.style.left=x+"px";
player.style.top=y+"px";

/* 吃糖果 */
candies.forEach((c,i)=>{
if(dist(player,c)<40){
c.remove();
candies.splice(i,1);
score++;
document.getElementById("score").innerText=score;

/* 星星特效 */
spark(c.style.left,c.style.top);

/* 解锁皇冠 */

crown.style.display="block";
crown.style.left=x+40+"px";
crown.style.top=y-60+"px";
}
}
});

/* 商店触发 */

shop.style.transform="scale(1.1)";

shop.style.transform="scale(1)";

requestAnimationFrame(loop);
}
loop();

   特效

function spark(x,y){
for(let i=0;i<6;i++){
let s=document.createElement("div");
s.style.position="absolute";
s.style.left=x;
s.style.top=y;
s.style.width="10px";
s.style.height="10px";
s.style.background="#fff";
s.style.borderRadius="50%";
s.style.opacity=1;
game.appendChild(s);

let vx=(Math.random()-0.5)*6;
let vy=(Math.random()-0.5)*6;

let t=setInterval(()=>{
let lx=parseFloat(s.style.left);
let ly=parseFloat(s.style.top);
s.style.left=lx+vx+"px";
s.style.top=ly+vy+"px";
s.style.opacity-=0.05;
if(s.style.opacity<=0){
clearInterval(t);
s.remove();
}
},16);
}
}

/* 距离 */
function dist(a,b){
let ax=parseInt(a.style.left);
let ay=parseInt(a.style.top);
let bx=parseInt(b.style.left);
let by=parseInt(b.style.top);
return Math.hypot(ax-bx,ay-by);
}

/* =========================
   触控摇杆（简化版）

let r=pad.getBoundingClientRect();

vx=(t.clientX-(r.left+r.width/2))/20;
vy=(t.clientY-(r.top+r.height/2))/20;

player.style.left=(parseInt(player.style.left)+vx)+"px";
player.style.top=(parseInt(player.style.top)+vy)+"px";
});
